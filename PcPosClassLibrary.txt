	public RecievedData SetSatus(byte[] header, string status)
	{
		RecievedData recievedData = new RecievedData();
		char[] array = status.ToCharArray();
		int num = 0;
		if (header[4] == 1 && header[5] == 119)
		{
			if (header[6] == 1 && header[7] == 112)
			{
				num = Convert.ToInt16(errorTypes.Payload);
				recievedData = recievedData.setError(num);
			}
			else if (header[6] == 1 && header[7] == 85)
			{
				num = Convert.ToInt16(errorTypes.Mac);
				recievedData = recievedData.setError(num);
			}
			else
			{
				num = Convert.ToInt16(errorTypes.Other);
				recievedData = recievedData.setError(num);
			}
		}
		else if (header[4] == 3 && header[5] == 99)
		{
			num = Convert.ToInt16(errorTypes2.CancelByPos);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '5' && array[1] == '0' && array[2] == '1')
		{
			num = Convert.ToInt16(errorTypes2.MacFail);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '0' && array[2] == '1')
		{
			num = Convert.ToInt16(errorTypes2.Cancel);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '0' && array[2] == '2')
		{
			num = Convert.ToInt16(errorTypes2.AlreadyReversed);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '0' && array[2] == '3')
		{
			num = Convert.ToInt16(errorTypes2.InvalidMerchant);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '0' && array[2] == '5')
		{
			num = Convert.ToInt16(errorTypes2.Cancel2);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '0' && array[2] == '6')
		{
			num = Convert.ToInt16(errorTypes2.InternalSystemError);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '0' && array[2] == '9')
		{
			num = Convert.ToInt16(errorTypes2.SystemBusy);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '0')
		{
			num = Convert.ToInt16(errorTypes2.PartialDispense);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '2')
		{
			num = Convert.ToInt16(errorTypes2.TransactionInvalid);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '3')
		{
			num = Convert.ToInt16(errorTypes2.IncorrectAmount);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '4')
		{
			num = Convert.ToInt16(errorTypes2.InvalidCardNumber);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '5')
		{
			num = Convert.ToInt16(errorTypes2.InvalidAcquire);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '7')
		{
			num = Convert.ToInt16(errorTypes2.UserCancel);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '1' && array[2] == '9')
		{
			num = Convert.ToInt16(errorTypes2.ReenterTransaction);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '2' && array[2] == '0')
		{
			num = Convert.ToInt16(errorTypes2.InvalidResponseCode);
			recievedData = setErrorTransactionStatus(num);
		}
		else if (array[0] == '0' && array[1] == '2' && array[2] == '1')
		{
			num = Convert.ToInt16(errorTypes2.NothingHappening);
			recievedData = setErrorTransactionStatus(num);
